cmake_minimum_required(VERSION 3.22)

find_package(ESMF 8.9.0 MODULE REQUIRED)

if(NOT DEFINED CMAKE_Fortran_COMPILER)
  set(CMAKE_Fortran_COMPILER "${ESMF_F90COMPILER}")
endif()

project(CompA
        VERSION 1.0
        LANGUAGES Fortran)

if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  set(CMAKE_Fortran_FLAGS "-fopt-info-all -foffload=nvptx-none -ffree-line-length-none -Wall -Wextra -Wconversion -Wno-unused -Wno-unused-dummy-argument")
  set(CMAKE_Fortran_FLAGS_RELEASE "-Ofast")
  set(CMAKE_Fortran_FLAGS_RELWITHDEBINFO "-g -fbacktrace -Ofast")
  set(CMAKE_Fortran_FLAGS_DEBUG "-g -fbacktrace -O0 -fcheck=all -ffpe-trap=invalid,zero,overflow,underflow")
elseif(CMAKE_Fortran_COMPILER_ID STREQUAL "IntelLLVM")
  set(CMAKE_Fortran_FLAGS_RELEASE "-Ofast")
  set(CMAKE_Fortran_FLAGS_RELWITHDEBINFO "-g -traceback -Ofast")
  set(CMAKE_Fortran_FLAGS_DEBUG "-g -traceback -O0 -check all -fpe0 -ftrapuv -init=snan,arrays")
elseif(CMAKE_Fortran_COMPILER_ID STREQUAL "NVHPC")
# The following option generates gpu code, and fall-back CPU code.
# There is a problem with default managed memory, leading to "cuMemFree"
# runtime errors (non fatal). But cleaner to switch to mem:separate mode.
  set(CMAKE_Fortran_FLAGS "-Minfo=all -stdpar=gpu -gpu=mem:separate")

# The following option generates gpu and multicore code at the same time.
# Turns out that the CPU code generated this way is less efficient than the
# fall-back CPU code generated by -stdpar=gpu. It supports threading through,
# yet turns out more efficient to decompose vs threading in this case.
#  set(CMAKE_Fortran_FLAGS "-Minfo=all -stdpar=multicore")
elseif(CMAKE_Fortran_COMPILER_ID STREQUAL "Cray")
  set(CMAKE_Fortran_FLAGS "-h acc -h offload_do_concurrent -h list=m")
else()
  message(WARNING "${CMAKE_Fortran_COMPILER_ID} Fortran compiler will be used with default options")
endif()

# Use STATIC library because SHARED can cause problems with memory management
#add_library(CompA SHARED compA.F90)
add_library(CompA STATIC compA.F90)

target_include_directories(CompA
  INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
)
target_link_libraries(CompA PUBLIC ESMF::ESMF)

option(USE_OPENACC "Enable OpenACC" OFF)
message(STATUS "Enable OpenACC: ${USE_OPENACC}")
if(USE_OPENACC)
  find_package(OpenACC REQUIRED Fortran)
  target_link_libraries(CompA PUBLIC OpenACC::OpenACC_Fortran)
endif()

option(USE_OPENMP "Enable OpenMP" OFF)
message(STATUS "Enable OpenMP: ${USE_OPENMP}")
if(USE_OPENMP)
  find_package(OpenMP REQUIRED Fortran)
  target_link_libraries(CompA PUBLIC OpenMP::OpenMP_Fortran)
endif()
